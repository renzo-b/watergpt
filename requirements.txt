# Runtime dependencies for the Entroply agent.

#   pip install -r requirements.txt
#
# Everything else the code imports (os, sys, json, re, pathlib, dataclasses,
# typing, argparse, base64, mimetypes, math) is in the Python standard library.

anthropic>=0.40      # Claude API client                 -> agent.py

# Retrieval. Needed to build or query the index; the agent loop itself runs
# without them, so an install that only wants to run evals can skip this block.
#
# Chroma rather than Postgres/pgvector: pgvector is a compiled extension with
# no prebuilt Windows package, so it costs a Docker daemon or an MSVC build
# chain before anything can be indexed. Chroma is a pip install and a
# directory (data/chroma, gitignored). Revisit if this ever outgrows a
# single-machine harness.
docling>=2.118       # PDF/docx -> DoclingDocument, HybridChunker -> scripts/inspect_parse.py, scripts/build_index.py
chromadb>=1.0        # embedded vector store                -> rag/store.py
openai>=1.40         # EMBEDDINGS ONLY - Anthropic has no embeddings endpoint,
                     # so vectors come from text-embedding-3. Every reasoning
                     # call in this repo is still Claude.  -> rag/embed.py
